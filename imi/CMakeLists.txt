# imi.cpp: an engine built for Seoul (LFM3-Omni) alone, from scratch: our own thread team, table layouts, kernels,
# step loops and pipeline. ggml is only the GGUF file reader (core/); brain.cpp is the first try on ggml's graph.
cmake_minimum_required(VERSION 3.18)
project(imi CXX C)
set(CMAKE_CXX_STANDARD 17)
set(GGML_DIR "" CACHE PATH "ggml source (the ggml/ folder of our llama.cpp fork)")
if (NOT GGML_DIR)
  message(FATAL_ERROR "pass -DGGML_DIR=<llama.cpp fork>/ggml")
endif()
# CPU only, like the church's Dell: no GPU, no vendor BLAS
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
# same CPU kernels and thread scheduler as the llama.cpp build we race against (Windows: ggml's own threads, so
# nunchi.exe needs no OpenMP DLL; imi has its own thread team either way)
if (WIN32)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
else()
  set(GGML_OPENMP ON CACHE BOOL "" FORCE)
endif()
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
# our kernels (and ggml's) are built for the CPU they run on (GGML_NATIVE)
option(IMI_NATIVE "build imi's own kernels and ggml's for this CPU" ON)
# A build for other computers (IMI_NATIVE OFF): IMI_ARCH names the oldest processors it runs on.
#   avx2: Intel Haswell (2013) / AMD Excavator (2015) and newer, the x86-64-v3 level spelled out feature by feature,
#         since GCC 10 (Ubuntu 22.04's Windows cross-compiler) does not know the level names
#   anything else is passed to -march as it is
set(IMI_ARCH "" CACHE STRING "processor level for a build for other computers (avx2)")
if (NOT IMI_NATIVE)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  if (IMI_ARCH STREQUAL "avx2")
    foreach (f GGML_SSE42 GGML_AVX GGML_AVX2 GGML_FMA GGML_F16C GGML_BMI2)
      set(${f} ON CACHE BOOL "" FORCE)
    endforeach()
  endif()
endif()
add_subdirectory(${GGML_DIR} ggml)
if (WIN32)
  # ggml-cpu.c opts its threads out of Windows 11's power saving with a part of Windows' API that old mingw-w64
  # headers lack (app/win_compat.h declares it where missing)
  set_source_files_properties(${GGML_DIR}/src/ggml-cpu/ggml-cpu.c TARGET_DIRECTORY ggml-cpu
                              PROPERTIES COMPILE_OPTIONS "-include;${CMAKE_CURRENT_SOURCE_DIR}/app/win_compat.h")
endif()
if (IMI_NATIVE AND NOT MSVC)
  add_compile_options(-march=native)
elseif (IMI_ARCH STREQUAL "avx2")
  add_compile_options(-march=x86-64 -mtune=generic -msse4.2 -mpopcnt -mavx -mavx2 -mfma -mf16c -mbmi -mbmi2 -mlzcnt -mmovbe)
elseif (IMI_ARCH)
  add_compile_options(-march=${IMI_ARCH})
endif()
add_executable(imi-gen gen.cpp model.cpp)
target_link_libraries(imi-gen PRIVATE ggml ggml-cpu)
add_executable(imi-brain-check brain_check.cpp brain.cpp model.cpp kernels/q4x8.cpp)
target_include_directories(imi-brain-check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(imi-brain-check PRIVATE ggml ggml-cpu)
# imi's own engine: our thread team, step loop, table layout and kernels (ggml is only the file reader here)
add_executable(imi-seoul-check core/seoul_check.cpp core/seoul.cpp core/tables.cpp model.cpp)
target_include_directories(imi-seoul-check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-seoul-check PRIVATE cxx_std_20)
target_link_libraries(imi-seoul-check PRIVATE ggml ggml-cpu)
add_executable(imi-voice-check core/voice_check.cpp core/voice.cpp core/tables.cpp model.cpp)
target_include_directories(imi-voice-check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-voice-check PRIVATE cxx_std_20)
target_link_libraries(imi-voice-check PRIVATE ggml ggml-cpu)
add_executable(imi-q8g-bench core/q8g_bench.cpp core/tables.cpp model.cpp)
target_include_directories(imi-q8g-bench PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(imi-q8g-bench PRIVATE ggml ggml-cpu)
add_executable(imi-codec-check core/codec_check.cpp core/codec.cpp core/tables.cpp model.cpp)
target_include_directories(imi-codec-check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-codec-check PRIVATE cxx_std_20)
target_link_libraries(imi-codec-check PRIVATE ggml ggml-cpu)
add_executable(imi-ears-check core/ears_check.cpp core/ears.cpp core/tables.cpp model.cpp)
target_include_directories(imi-ears-check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-ears-check PRIVATE cxx_std_20)
target_link_libraries(imi-ears-check PRIVATE ggml ggml-cpu)
add_executable(imi-words-check core/words_check.cpp core/words.cpp model.cpp)
target_include_directories(imi-words-check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-words-check PRIVATE cxx_std_20)
target_link_libraries(imi-words-check PRIVATE ggml ggml-cpu)
# imi hears: ears + brain, greedy Korean words
add_executable(imi-hear core/hear.cpp core/ears.cpp core/seoul.cpp core/words.cpp core/tables.cpp model.cpp)
target_include_directories(imi-hear PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-hear PRIVATE cxx_std_20)
target_link_libraries(imi-hear PRIVATE ggml ggml-cpu)
# research: the live hearing layout (LFM3-Audio's) on Seoul, 12 rows (0.96 s) at a time
add_executable(imi-hear-live core/hear_live.cpp core/ears.cpp core/seoul.cpp core/words.cpp core/tables.cpp model.cpp)
target_include_directories(imi-hear-live PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-hear-live PRIVATE cxx_std_20)
target_link_libraries(imi-hear-live PRIVATE ggml ggml-cpu)
# research: hearing short pieces cut at the speaker's pauses, one process
add_executable(imi-hear-pieces core/hear_pieces.cpp core/ears.cpp core/seoul.cpp core/words.cpp core/tables.cpp model.cpp)
target_include_directories(imi-hear-pieces PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-hear-pieces PRIVATE cxx_std_20)
target_link_libraries(imi-hear-pieces PRIVATE ggml ggml-cpu)
# research: translating clause by clause as the Korean comes in, with the clauses before as context
add_executable(imi-translate-pieces core/translate_pieces.cpp core/seoul.cpp core/words.cpp core/tables.cpp model.cpp)
target_include_directories(imi-translate-pieces PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-translate-pieces PRIVATE cxx_std_20)
target_link_libraries(imi-translate-pieces PRIVATE ggml ggml-cpu)
# the pipeline: hear -> translate (-> speak), one brain, one thread team
add_executable(imi-nunchi core/nunchi.cpp core/ears.cpp core/seoul.cpp core/words.cpp core/talk.cpp core/voice.cpp core/codec.cpp core/tables.cpp model.cpp)
target_include_directories(imi-nunchi PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-nunchi PRIVATE cxx_std_20)
target_link_libraries(imi-nunchi PRIVATE ggml ggml-cpu)
# the live pipeline: speak window N while hearing and translating N+1, word steps riding voice frames
add_executable(imi-flow core/flow.cpp core/ears.cpp core/seoul.cpp core/words.cpp core/talk.cpp core/voice.cpp core/codec.cpp core/tables.cpp model.cpp)
target_include_directories(imi-flow PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-flow PRIVATE cxx_std_20)
target_link_libraries(imi-flow PRIVATE ggml ggml-cpu)
# research: the instant interpreter on a clock (pieces cut at pauses, sentence-end translating, voice per sentence)
add_executable(imi-instant core/instant.cpp core/ears.cpp core/seoul.cpp core/words.cpp core/talk.cpp core/voice.cpp core/codec.cpp core/tables.cpp model.cpp)
target_include_directories(imi-instant PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-instant PRIVATE cxx_std_20)
target_link_libraries(imi-instant PRIVATE ggml ggml-cpu)
# imi speaks: brain + voice generator + codec, one thread team
add_executable(imi-speak core/speak.cpp core/talk.cpp core/seoul.cpp core/voice.cpp core/codec.cpp core/words.cpp core/tables.cpp model.cpp)
target_include_directories(imi-speak PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-speak PRIVATE cxx_std_20)
target_link_libraries(imi-speak PRIVATE ggml ggml-cpu)
# research: are the ears' rows final while the audio is still coming in?
add_executable(imi-ears-causal core/ears_causal.cpp core/ears.cpp core/tables.cpp model.cpp)
target_include_directories(imi-ears-causal PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(imi-ears-causal PRIVATE cxx_std_20)
target_link_libraries(imi-ears-causal PRIVATE ggml ggml-cpu)
# the app: one program for the room's computer (the live interpreter, its pages, every phone's captions and voice)
add_executable(nunchi app/main.cpp app/live.cpp app/server.cpp app/qr.cpp app/net.cpp core/ears.cpp core/seoul.cpp core/words.cpp core/talk.cpp core/voice.cpp core/codec.cpp core/tables.cpp model.cpp app/cpu_check.cpp)
if (IMI_ARCH STREQUAL "avx2")
  # the processor check runs before anything else and must itself run on any 64-bit processor
  target_compile_definitions(nunchi PRIVATE NUNCHI_NEEDS_AVX2)
  set_source_files_properties(app/cpu_check.cpp PROPERTIES COMPILE_OPTIONS "-mno-avx;-mno-avx2;-mno-fma;-mno-f16c;-mno-bmi;-mno-bmi2")
endif()
target_include_directories(nunchi PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(nunchi PRIVATE cxx_std_20)
target_link_libraries(nunchi PRIVATE ggml ggml-cpu)
if (WIN32)
  # Windows 10 is the oldest Nunchi runs on; without this mingw-w64 assumes XP and hides newer calls (inet_ntop)
  target_compile_definitions(nunchi PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(nunchi PRIVATE ws2_32 iphlpapi shell32 -static -static-libgcc -static-libstdc++)
endif()
